## Minimal repository fit

```ts
// apps/api/src/modules/identity/auth.ts
import { drizzleAdapter } from "@better-auth/drizzle-adapter";
import { betterAuth } from "better-auth";
import { drizzle } from "drizzle-orm/neon-http";
import { getSqlClient } from "../../platform/database.js";
import * as schema from "@veyra/db";

const sql = getSqlClient();
if (!sql) throw new Error("Better Auth requires DATABASE_URL.");

const db = drizzle({ client: sql, schema });

export const auth = betterAuth({
  database: drizzleAdapter(db, {
    provider: "pg",
    schema,
    usePlural: true,
  }),
  baseURL: environment.VEYRA_API_ORIGIN,
  trustedOrigins: [environment.VEYRA_APP_ORIGIN],
  emailAndPassword: { enabled: true },
});
```
